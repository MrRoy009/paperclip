cmake_minimum_required(VERSION 3.20)
project(fnq LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
 add_compile_options(/W4 /WX)
else()
 add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()
add_library(fnq_core STATIC src/core.cpp)
target_include_directories(fnq_core PUBLIC include)
enable_testing()
foreach(area arithmetic sizing events ledger)
 add_executable(test_${area} tests/${area}.cpp)
 target_link_libraries(test_${area} PRIVATE fnq_core)
endforeach()
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/docs/engineer-milestone-1.md" cases REGEX "^\\| `T[0-9][0-9]_")
foreach(row IN LISTS cases)
 string(REGEX MATCH "T[0-9][0-9]_[a-z_]+" name "${row}")
 string(SUBSTRING "${name}" 1 2 number)
 if(number STREQUAL "25")
  add_test(NAME ${name} COMMAND ${CMAKE_COMMAND} -DROOT=${CMAKE_CURRENT_SOURCE_DIR} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/no_float.cmake)
 elseif(number LESS 7 OR number EQUAL 12 OR (number GREATER 14 AND number LESS 19) OR number EQUAL 24 OR number GREATER 28)
  add_test(NAME ${name} COMMAND test_ledger ${number})
 elseif(number LESS 12)
  add_test(NAME ${name} COMMAND test_arithmetic ${number})
 elseif(number GREATER 18 AND number LESS 24)
  add_test(NAME ${name} COMMAND test_events ${number})
 else()
  add_test(NAME ${name} COMMAND test_sizing ${number})
 endif()
endforeach()

# Independent-QA regression: no backward fill time; same-event partials remain valid.
add_test(NAME R01_fill_event_monotonic COMMAND test_ledger 31)
