cmake_minimum_required(VERSION 3.20)
project(fnq_qa_strategy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(fnq_under_audit STATIC ${ROOT}/src/core.cpp ${ROOT}/src/bars.cpp ${ROOT}/src/features.cpp ${ROOT}/src/orders.cpp ${ROOT}/src/strategy.cpp)
target_include_directories(fnq_under_audit PUBLIC ${ROOT}/include)
enable_testing()
add_executable(qa_probe probes.cpp)
target_link_libraries(qa_probe PRIVATE fnq_under_audit)
foreach(n RANGE 1 24)
 add_test(NAME Q${n} COMMAND qa_probe ${n})
endforeach()
